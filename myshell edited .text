//Femto shell//

#include <stdio.h>
#include <string.h>

int main(int argc, char *argv[])
{
    char input[1024];


    while (1)
    {
        printf("Abdelghafar's Femto Shell:~$ ");

        if (fgets(input, sizeof(input), stdin) == NULL)
        {
            break;
        }

        input[strcspn(input, "\n")] = '\0';

        if (strlen(input) == 0)
        {
            continue;
        }

        if (strcmp(input, "exit") == 0)
        {
            printf("Good Bye :)\n");
            break;
        }
        else if (strncmp(input, "echo", 4) == 0 &&
                 (input[4] == ' ' || input[4] == '\0'))
        {
            if (input[4] == '\0')
            {
                printf("\n");
            }
            else
            {
                printf("%s\n", input + 5);
            }
        }
        else
        {
            printf("Invalid command\n");
        }
    }

    return 0;
}


//Pico shell//

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <limits.h>
#include <sys/types.h>
#include <sys/wait.h>

#define MAX_LINE 1024
#define MAX_ARGS 100

int echo(int argc, char *argv[])
{
    int i;

    for (i = 1; i < argc; i++)
    {
        printf("%s", argv[i]);

        if (i < argc - 1)
            printf(" ");
    }

    printf("\n");
    return 0;
}

int pwd(void)
{
    char cwd[PATH_MAX];

    if (getcwd(cwd, sizeof(cwd)) == NULL)
    {
        perror("getcwd");
        return 1;
    }

    printf("%s\n", cwd);
    return 0;
}

int cd(int argc, char *argv[])
{
    if (argc < 2)
    {
        printf("cd: missing argument\n");
        return 1;
    }

    if (chdir(argv[1]) != 0)
    {
        perror("cd");
        return 1;
    }

    return 0;
}

int main(void)
{
    char line[MAX_LINE];
    char *argv[MAX_ARGS];
    int argc;

    while (1)
    {
        printf("Abdelghafar's Pico Shell:~$ ");

        if (fgets(line, sizeof(line), stdin) == NULL)
            break;

        line[strcspn(line, "\n")] = '\0';

        argc = 0;

        argv[argc] = strtok(line, " ");

        while (argv[argc] != NULL)
        {
            argc++;
            argv[argc] = strtok(NULL, " ");
        }

        if (argc == 0)
            continue;
        if (strcmp(argv[0], "echo") == 0)
        {
            echo(argc, argv);
        }
        else if (strcmp(argv[0], "pwd") == 0)
        {
            pwd();
        }
        else if (strcmp(argv[0], "cd") == 0)
        {
            cd(argc, argv);
        }
        else if (strcmp(argv[0], "exit") == 0)
        {
            printf("Good Bye :)\n");
            break;
        }
        else
        {
            pid_t pid = fork();

            if (pid < 0)
            {
                perror("fork");
            }
            else if (pid == 0)
            {
                execvp(argv[0], argv);
                perror("execvp");
                exit(1);
            }
            else
            {
                wait(NULL);
            }
        }
    }

    return 0;
}
