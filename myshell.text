//Femto shell//

#include <stdio.h>
#include <string.h>

int main(int argc, char *argv[])
{
    char input[1024];


    while (1)
    {
        printf("Abdelghafar's Femto Shell:~$ ");

        if (fgets(input, sizeof(input), stdin) == NULL)
        {
            break;
        }

        input[strcspn(input, "\n")] = '\0';

        if (strlen(input) == 0)
        {
            continue;
        }

        if (strcmp(input, "exit") == 0)
        {
            printf("Good Bye :)\n");
            break;
        }
        else if (strncmp(input, "echo", 4) == 0 &&
                 (input[4] == ' ' || input[4] == '\0'))
        {
            if (input[4] == '\0')
            {
                printf("\n");
            }
            else
            {
                printf("%s\n", input + 5);
            }
        }
        else
        {
            printf("Invalid command\n");
        }
    }

    return 0;
}


//Pico shell//

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>

#define MAX_LINE 1024

int main()
{
    char line[MAX_LINE];

    while (1) {
	printf("Abdelghafar's Pico Shell:~$ ");
	fflush(stdout);

	if (fgets(line, MAX_LINE, stdin) == NULL) {
	    printf("\n");
	    break;
	}

	line[strcspn(line, "\n")] = '\0';

	int capacity = 10;
	int argc = 0;
	char **argv = (char **) malloc(capacity * sizeof(char *));
	if (argv == NULL) {
	    perror("malloc");
	    continue;
	}

	char *token = strtok(line, " ");

	while (token != NULL) {
	    if (argc >= capacity) {
		capacity *= 2;
		char **temp =
		    (char **) realloc(argv, capacity * sizeof(char *));
		if (temp == NULL) {
		    perror("realloc");
		    free(argv);
		    argv = NULL;
		    break;
		}
		argv = temp;
	    }

	    argv[argc++] = token;
	    token = strtok(NULL, " ");
	}

	if (argv == NULL)
	    continue;

	argv[argc] = NULL;

	if (argc == 0) {
	    free(argv);
	    continue;
	}

	if (strcmp(argv[0], "exit") == 0) {
	    printf("Good Bye :)\n");
	    free(argv);
	    break;
	}

	else if (strcmp(argv[0], "echo") == 0) {
	    for (int i = 1; i < argc; i++) {
		printf("%s", argv[i]);
		if (i < argc - 1)
		    printf(" ");
	    }
	    printf("\n");
	}

	else if (strcmp(argv[0], "pwd") == 0) {
	    char cwd[1024];

	    if (getcwd(cwd, sizeof(cwd)) != NULL)
		printf("%s\n", cwd);
	    else
		perror("getcwd");
	}

	else if (strcmp(argv[0], "cd") == 0) {
	    if (argc < 2) {
		fprintf(stderr, "cd: missing operand\n");
	    } else {
		if (chdir(argv[1]) != 0)
		    perror("cd");
	    }
	}

	else {
	    pid_t pid = fork();

	    if (pid < 0) {
		perror("fork");
	    } else if (pid == 0) {
		execvp(argv[0], argv);

		perror("execvp");
		exit(EXIT_FAILURE);
	    } else {
		wait(NULL);
	    }
	}

	free(argv);
    }

    return 0;
}
